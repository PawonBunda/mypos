let videoStream = null; let detector = null; async function openCamera(){ document.getElementById("cameraModal").style.display = "flex"; const video = document.getElementById("video"); try { videoStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" } }); video.srcObject = videoStream; video.play(); // Barcode Detector (Chrome Android) if ('BarcodeDetector' in window) { detector = new BarcodeDetector({ formats: ['ean_13','code_128','qr_code'] }); scanBarcode(); } else { showToast("Browser tidak support scanner","error"); } } catch (err){ showToast("Gagal akses kamera","error"); } } function closeCamera(){ document.getElementById("cameraModal").style.display = "none"; if(videoStream){ videoStream.getTracks().forEach(t => t.stop()); } } async function scanBarcode(){ const video = document.getElementById("video"); if(!detector) return; try{ const barcodes = await detector.detect(video); if(barcodes.length > 0){ let kode = barcodes[0].rawValue; barcodeBaru.value = kode; closeCamera(); autoDetectBarcode(); // langsung isi form return; } } catch(e){} requestAnimationFrame(scanBarcode); }
